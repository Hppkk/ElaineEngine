layout(location = 0) in vec4 inColor;
layout(location = 1) in vec3 inUV;

layout(location = 0) out vec4 outColor;
layout(set = 2, binding = 0) uniform samplerCube S_Texture0;

void main()
{
    outColor = inColor;
    outColor.x = 0.0;
    outColor.y = 1.0;
    outColor.z = 0.0;
    outColor.w = 1.0;
    outColor = gl_FrontFacing ? vec4(0, 1, 0, 1) : vec4(1, 0, 0, 1);
    outColor.rgb = texture(S_Texture0, inUV).rgb;
}
